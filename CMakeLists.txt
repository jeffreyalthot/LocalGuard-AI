cmake_minimum_required(VERSION 3.16)
project(LocalGuard LANGUAGES CXX)
if(NOT WIN32)
 message(FATAL_ERROR "Windows required")
endif()
add_executable(LocalGuard main.cpp Monitor.cpp Detector.cpp Events.cpp)
target_compile_features(LocalGuard PRIVATE cxx_std_17)
target_compile_definitions(LocalGuard PRIVATE _WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(LocalGuard PRIVATE iphlpapi ws2_32 advapi32)
